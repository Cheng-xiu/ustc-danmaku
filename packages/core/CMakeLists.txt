# 唯一源码清单同时供原生、Wasm 和交付脚本消费。
set(DEMO_CORE_ROOT "${CMAKE_CURRENT_SOURCE_DIR}")
file(STRINGS "${DEMO_CORE_ROOT}/sources.txt" DEMO_CORE_RELATIVE)
list(LENGTH DEMO_CORE_RELATIVE DEMO_CORE_COUNT)
if(NOT DEMO_CORE_COUNT EQUAL 16)
    message(FATAL_ERROR "Expected the shared 16-file C/AI source list")
endif()
set(DEMO_CORE_SOURCES "")
foreach(DEMO_SOURCE IN LISTS DEMO_CORE_RELATIVE)
    if(NOT DEMO_SOURCE MATCHES "^(core|ai)/[A-Za-z0-9_]+[.]c$")
        message(FATAL_ERROR "Invalid shared core source: ${DEMO_SOURCE}")
    endif()
    list(APPEND DEMO_CORE_SOURCES "${DEMO_CORE_ROOT}/${DEMO_SOURCE}")
endforeach()
set(DEMO_CORE_INCLUDE_DIRS "${DEMO_CORE_ROOT}/core" "${DEMO_CORE_ROOT}/ai")
add_library(demo_core STATIC ${DEMO_CORE_SOURCES})
set_source_files_properties(${DEMO_CORE_SOURCES} PROPERTIES LANGUAGE C)
target_include_directories(demo_core PUBLIC ${DEMO_CORE_INCLUDE_DIRS})
target_compile_definitions(demo_core PUBLIC DEMO_CORE_BUILD=1)
if(NOT MSVC)
    target_link_libraries(demo_core PUBLIC m)
endif()
# 父入口的窄测试与 Wasm 共享同一个绝对路径清单。
set(DEMO_CORE_ROOT "${DEMO_CORE_ROOT}" PARENT_SCOPE)
set(DEMO_CORE_SOURCES "${DEMO_CORE_SOURCES}" PARENT_SCOPE)
set(DEMO_CORE_INCLUDE_DIRS "${DEMO_CORE_INCLUDE_DIRS}" PARENT_SCOPE)
