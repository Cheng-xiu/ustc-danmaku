cmake_minimum_required(VERSION 3.16)
project(demo_web_core C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")
# 可独立从此包配置构建；核心库始终由同一个包和 sources.txt 提供。
get_filename_component(DEMO_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
if(NOT TARGET demo_core)
    add_subdirectory("${DEMO_ROOT}/packages/core" "${CMAKE_CURRENT_BINARY_DIR}/core")
endif()
if(EMSCRIPTEN)
    target_compile_options(demo_core PRIVATE -O2 -Wall -Wextra)
    add_executable(demo-core demo_bridge.c)
    target_link_libraries(demo-core PRIVATE demo_core)
    target_compile_options(demo-core PRIVATE -O2 -Wall -Wextra)
    target_link_options(demo-core PRIVATE --no-entry -O2
        "-sMODULARIZE=1" "-sEXPORT_ES6=1" "-sENVIRONMENT=web,node"
        "-sFILESYSTEM=0" "-sALLOW_MEMORY_GROWTH=1" "-sSTACK_SIZE=1048576"
        "-sINITIAL_MEMORY=16777216" "-sMAXIMUM_MEMORY=67108864"
        "-sEXPORTED_FUNCTIONS=['_demo_reset','_demo_reset_sized','_demo_step','_demo_step_aim','_demo_preview','_demo_preview_size','_demo_snapshot','_demo_snapshot_size','_demo_dispose']"
        "-sEXPORTED_RUNTIME_METHODS=['HEAPU8']")
    set_target_properties(demo-core PROPERTIES SUFFIX ".mjs")
else()
    add_library(demo_web_bridge STATIC demo_bridge.c)
    target_link_libraries(demo_web_bridge PUBLIC demo_core)
    target_include_directories(demo_web_bridge PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
    if(MSVC)
        target_compile_options(demo_core PRIVATE /W4 /utf-8)
        target_compile_definitions(demo_core PRIVATE _CRT_SECURE_NO_WARNINGS)
        target_compile_options(demo_web_bridge PRIVATE /W4 /utf-8)
    else()
        target_compile_options(demo_core PRIVATE -Wall -Wextra)
        target_compile_options(demo_web_bridge PRIVATE -Wall -Wextra)
    endif()
    add_executable(demo_bridge_probe bridge_probe.c)
    target_link_libraries(demo_bridge_probe PRIVATE demo_web_bridge)
    enable_testing()
    add_test(NAME demo_bridge_probe COMMAND demo_bridge_probe)
endif()
