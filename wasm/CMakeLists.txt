cmake_minimum_required(VERSION 3.16)
project(demo_web_core C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
get_filename_component(DEMO_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/.." ABSOLUTE)
# Consume the root source list; never fork the native and Wasm core lists.
file(READ "${DEMO_ROOT}/CMakeLists.txt" ROOT_CMAKE)
string(REGEX MATCH "set\\(DEMO_CORE_SOURCES[^)]*\\)" SOURCE_BLOCK "${ROOT_CMAKE}")
string(REGEX MATCHALL "(core|ai)/[A-Za-z0-9_]+\\.c" CORE_RELATIVE "${SOURCE_BLOCK}")
list(LENGTH CORE_RELATIVE CORE_COUNT)
if(NOT CORE_COUNT EQUAL 15)
    message(FATAL_ERROR "Expected the frozen 15-file core/ai source list")
endif()
set(CORE_ABSOLUTE "")
foreach(SOURCE IN LISTS CORE_RELATIVE)
    list(APPEND CORE_ABSOLUTE "${DEMO_ROOT}/${SOURCE}")
endforeach()
if(EMSCRIPTEN)
    add_executable(demo-core ${CORE_ABSOLUTE} demo_bridge.c)
    target_include_directories(demo-core PRIVATE "${DEMO_ROOT}/core" "${DEMO_ROOT}/ai")
    target_compile_options(demo-core PRIVATE -O2 -Wall -Wextra)
    target_link_options(demo-core PRIVATE --no-entry -O2
        "-sMODULARIZE=1" "-sEXPORT_ES6=1" "-sENVIRONMENT=web,node"
        "-sFILESYSTEM=0" "-sALLOW_MEMORY_GROWTH=1" "-sSTACK_SIZE=1048576"
        "-sINITIAL_MEMORY=16777216" "-sMAXIMUM_MEMORY=67108864"
        "-sEXPORTED_FUNCTIONS=['_demo_reset','_demo_reset_sized','_demo_step','_demo_snapshot','_demo_snapshot_size','_demo_dispose']"
        "-sEXPORTED_RUNTIME_METHODS=['HEAPU8']")
    target_link_libraries(demo-core PRIVATE m)
    set_target_properties(demo-core PROPERTIES SUFFIX ".mjs")
else()
    add_library(demo_web_bridge STATIC ${CORE_ABSOLUTE} demo_bridge.c)
    target_include_directories(demo_web_bridge PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}"
        "${DEMO_ROOT}/core" "${DEMO_ROOT}/ai")
    if(MSVC)
        target_compile_options(demo_web_bridge PRIVATE /W4 /utf-8)
    else()
        target_compile_options(demo_web_bridge PRIVATE -Wall -Wextra)
        target_link_libraries(demo_web_bridge PUBLIC m)
    endif()
    add_executable(demo_bridge_probe bridge_probe.c)
    target_link_libraries(demo_bridge_probe PRIVATE demo_web_bridge)
    enable_testing()
    add_test(NAME demo_bridge_probe COMMAND demo_bridge_probe)
endif()
